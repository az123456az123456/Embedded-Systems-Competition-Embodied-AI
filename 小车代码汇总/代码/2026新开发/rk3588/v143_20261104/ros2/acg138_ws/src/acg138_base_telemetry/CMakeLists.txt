cmake_minimum_required(VERSION 3.16)
project(acg138_base_telemetry LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(acg138_telemetry_decode STATIC vendor/imu_data.c vendor/sensor_data.c)
target_include_directories(acg138_telemetry_decode PUBLIC vendor)
target_link_libraries(acg138_telemetry_decode m)
add_executable(base_telemetry_node src/base_telemetry_node.cpp)
target_include_directories(base_telemetry_node PRIVATE include vendor)
ament_target_dependencies(base_telemetry_node rclcpp nav_msgs sensor_msgs std_msgs geometry_msgs tf2_ros)
target_link_libraries(base_telemetry_node acg138_telemetry_decode)
add_executable(odom_health_gate src/odom_health_gate.cpp)
ament_target_dependencies(odom_health_gate rclcpp nav_msgs geometry_msgs tf2_ros)
install(TARGETS odom_health_gate DESTINATION lib/${PROJECT_NAME})
install(TARGETS base_telemetry_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_odometry tests/test_odometry.cpp)
 target_include_directories(test_odometry PRIVATE include vendor)
 target_link_libraries(test_odometry acg138_telemetry_decode)
 add_test(NAME odometry_protocol COMMAND test_odometry)
 add_executable(test_kernel_timestamp tests/test_kernel_timestamp.cpp)
 target_include_directories(test_kernel_timestamp PRIVATE include)
 target_link_libraries(test_kernel_timestamp pthread)
 add_test(NAME kernel_timestamp COMMAND test_kernel_timestamp)
endif()
ament_package()
