cmake_minimum_required(VERSION 3.16)
project(acg138_rplidar VERSION 0.37.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
set(SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sl_lidar_sdk")
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS
  "${SDK_DIR}/src/*.cpp" "${SDK_DIR}/src/arch/linux/*.cpp"
  "${SDK_DIR}/src/hal/*.cpp" "${SDK_DIR}/src/dataunpacker/*.cpp"
  "${SDK_DIR}/src/dataunpacker/unpacker/*.cpp")
add_library(c1_sl_lidar_sdk STATIC ${SDK_SOURCES})
target_include_directories(c1_sl_lidar_sdk SYSTEM PUBLIC "${SDK_DIR}/include")
target_include_directories(c1_sl_lidar_sdk PRIVATE "${SDK_DIR}/src")
target_link_libraries(c1_sl_lidar_sdk PUBLIC Threads::Threads)

option(ACG138_CORE_ONLY "Build SDK and scan conversion tests without ROS 2" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(c1_scan_core_test tests/scan_core_test.cpp)
  target_include_directories(c1_scan_core_test PRIVATE include)
  target_compile_options(c1_scan_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME c1_scan_core COMMAND c1_scan_core_test)
endif()
if(ACG138_CORE_ONLY)
  return()
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(c1_node src/c1_node.cpp)
target_include_directories(c1_node PRIVATE include)
target_compile_options(c1_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(c1_node c1_sl_lidar_sdk)
ament_target_dependencies(c1_node rclcpp sensor_msgs)
install(TARGETS c1_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES vendor/LICENSE-Slamtec.txt vendor/provenance.json DESTINATION share/${PROJECT_NAME}/vendor)
ament_package()
