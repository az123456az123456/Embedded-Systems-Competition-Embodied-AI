cmake_minimum_required(VERSION 3.16)
project(exploration_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "ASan UBSan" OFF)
if(SANITIZE)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -fno-pie)
add_link_options(-fsanitize=address,undefined -no-pie)
endif()
enable_testing()
set(P ../acg138_fpga_map_bridge)
add_library(wire ${P}/src/nav3_wire.cpp ${P}/src/nav3_ethernet.cpp ${P}/src/nav2_wire.cpp)
target_include_directories(wire PUBLIC ${P}/include)
set(R ${P}/tests/reference/nav3)
add_library(oracle ${R}/nav_exec_wire.c ${R}/candidate26/softcore/firmware/nav_exec.c ${R}/candidate26/softcore/firmware/nav_exec_protocol.c ${R}/candidate26/softcore/firmware/nav_protocol.c ${R}/candidate26/softcore/firmware/astar.c)
target_compile_options(wire PRIVATE -Wall -Wextra -Wpedantic)
foreach(suite execution plan stream map)
add_executable(test_${suite} ${P}/tests/test_nav3_${suite}.cpp)
target_link_libraries(test_${suite} wire oracle)
add_test(NAME ${suite} COMMAND test_${suite} ${CMAKE_CURRENT_BINARY_DIR}/results/${suite})
endforeach()
add_library(nav_authority ../gateway/nav_authority.cc ../gateway/fpga_delegation.cc ../gateway/nva_socket.cc)
target_link_libraries(nav_authority PUBLIC wire oracle)
add_executable(test_authority ../gateway/test_authority.cpp)
target_link_libraries(test_authority nav_authority)
add_test(NAME authority COMMAND test_authority ${CMAKE_CURRENT_BINARY_DIR}/results/authority)
add_executable(test_delegation ../gateway/test_delegation.cpp)
target_link_libraries(test_delegation nav_authority)
add_test(NAME delegation COMMAND test_delegation ${CMAKE_CURRENT_BINARY_DIR}/results/delegation)
