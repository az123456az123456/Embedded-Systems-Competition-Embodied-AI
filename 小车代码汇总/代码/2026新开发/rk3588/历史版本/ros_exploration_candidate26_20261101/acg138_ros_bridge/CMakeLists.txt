cmake_minimum_required(VERSION 3.16)
project(acg138_ros_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(acg138_nav_interfaces REQUIRED)
find_package(Threads REQUIRED)
set(COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../common)
add_executable(ros_bridge_node src/bridge_node.cpp ${COMMON}/ros_bridge_codec.cc ${COMMON}/ros_bridge_session.cc ${COMMON}/ros_bridge_transport.cc)
target_include_directories(ros_bridge_node PRIVATE ${COMMON})
target_compile_options(ros_bridge_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(ros_bridge_node rclcpp std_msgs std_srvs nav_msgs sensor_msgs geometry_msgs acg138_nav_interfaces)
target_link_libraries(ros_bridge_node Threads::Threads)
install(TARGETS ros_bridge_node DESTINATION lib/${PROJECT_NAME})
ament_package()
