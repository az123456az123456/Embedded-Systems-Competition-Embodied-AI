cmake_minimum_required(VERSION 3.16)
project(acg138_fpga_map_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(acg138_nav_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Threads REQUIRED)
add_library(acg138_map_core src/map_core.cpp src/nav2_wire.cpp src/raw_ethernet.cpp)
target_include_directories(acg138_map_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(acg138_map_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(acg138_map_core PUBLIC Threads::Threads)
add_executable(fpga_map_bridge src/bridge_node.cpp)
ament_target_dependencies(fpga_map_bridge acg138_nav_interfaces rclcpp nav_msgs geometry_msgs std_msgs std_srvs tf2 tf2_ros)
target_link_libraries(fpga_map_bridge acg138_map_core)
install(TARGETS fpga_map_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_executable(test_navigation_heading tests/test_navigation_heading.cpp)
  target_include_directories(test_navigation_heading PRIVATE include)
  add_test(NAME navigation_heading COMMAND test_navigation_heading)
  add_executable(test_footprint_inflation tests/test_footprint_inflation.cpp)
  target_link_libraries(test_footprint_inflation acg138_map_core)
  add_test(NAME footprint_inflation COMMAND test_footprint_inflation)
  add_executable(test_nav3_map_latch tests/test_nav3_map_latch.cpp)
  target_include_directories(test_nav3_map_latch PRIVATE include)
  add_test(NAME nav3_manual_map_latch COMMAND test_nav3_map_latch)
  add_test(NAME map_bridge_host_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_host_tests.py)
endif()
include(cmake/Nav3Stage0.cmake)
include(cmake/Nav3Stage1.cmake)
include(cmake/Nav3Stage2.cmake)
ament_package()

include(cmake/Nav3Stage3.cmake)
