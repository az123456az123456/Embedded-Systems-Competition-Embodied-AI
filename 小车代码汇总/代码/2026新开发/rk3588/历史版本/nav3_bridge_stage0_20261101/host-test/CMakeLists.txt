cmake_minimum_required(VERSION 3.16)
project(nav3_stage0_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NAV3_SANITIZE "Address and undefined-behavior sanitizers" OFF)
set(BUILD_TESTING ON)
enable_testing()
if(NAV3_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_subdirectory(../package package)
